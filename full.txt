============================================================
 Question 1: FCFS - First Come First Serve
============================================================
Description:
 - Processes run in arrival order.
 - Simple but may cause the Convoy Effect.

Source Code (C++):

#include<iostream>
#include<string>
using namespace std;
int main(){
    cout << "============================" << endl;
    cout << " Mohd Anas " << endl;
    cout << " Roll : 24BCS061 " << endl;
    cout << " FCFS CPU Scheduling " << endl;
    cout << "============================" << endl;
    int n;
    cout << "\nEnter number of processes: ";
    cin >> n;
    string name[n];
    int at[n], bt[n], ct[n], tat[n], wt[n];
    for(int i = 0; i < n; i++){
        name[i] = "P" + to_string(i+1);
        cout << "Enter Arrival Time for P" << (i+1) << ": "; cin >> at[i];
        cout << "Enter Burst Time for P" << (i+1) << ": "; cin >> bt[i];
    }
    for(int i = 0; i < n-1; i++)
        for(int j = 0; j < n-i-1; j++)
            if(at[j] > at[j+1]){
                swap(at[j],at[j+1]);
                swap(bt[j],bt[j+1]);
                swap(name[j],name[j+1]);
            }
    int gantt[200], gs[200], ge[200], gc = 0;
    int cur = 0;
    for(int i = 0; i < n; i++){
        if(cur < at[i]) cur = at[i];
        gs[gc]=cur; gantt[gc]=i; ge[gc]=cur+bt[i]; gc++;
        cur += bt[i];
        ct[i] = cur;
        tat[i] = ct[i] - at[i];
        wt[i] = tat[i] - bt[i];
    }
    cout << "\n--- Gantt Chart ---\n|";
    for(int i=0;i<gc;i++) cout << " " << name[gantt[i]] << " |";
    cout << "\n" << gs[0];
    for(int i=0;i<gc;i++) cout << "----" << ge[i];
    cout << "\n";
    cout << "\nProcess AT BT CT TAT WT\n";
    cout << "--------------------------------\n";
    float ttat=0, twt=0;
    for(int i=0;i<n;i++){
        cout << name[i] <<" "<<at[i]<<" "<<bt[i]<<" "
             <<ct[i]<<" "<<tat[i]<<" "<<wt[i]<<"\n";
        ttat+=tat[i]; twt+=wt[i];
    }
    cout << "--------------------------------\n";
    cout << "Average TAT : " << ttat/n << "\n";
    cout << "Average WT  : " << twt/n << "\n";
    return 0;
}

 Question 3: SRTF - Shortest Remaining Time First (Preemptive)
============================================================

============================================================
 Question 6: Round Robin Scheduling
============================================================
Description:
 - Each process gets a fixed Time Quantum.
 - If not done it goes back to the end of the queue.
 - Fair and widely used in time-sharing systems.

Source Code (C++):

#include<iostream>
#include<string>
#include<queue>
using namespace std;
int main(){
    cout << "============================" << endl;
    cout << " Mohd Anas " << endl;
    cout << " Roll : 24BCS061 " << endl;
    cout << " Round Robin Scheduling " << endl;
    cout << "============================" << endl;
    int n, tq;
    cout << "\nEnter number of processes: "; cin >> n;
    cout << "Enter Time Quantum : "; cin >> tq;
    string name[n];
    int at[n], bt[n], rem[n], ct[n], tat[n], wt[n];
    for(int i=0;i<n;i++){
        name[i]="P"+to_string(i+1);
        cout<<"Enter Arrival Time for P"<<(i+1)<<": "; cin>>at[i];
        cout<<"Enter Burst Time for P"<<(i+1)<<": "; cin>>bt[i];
        rem[i]=bt[i];
    }
    int gantt[1000], gs[1000], ge[1000], gc=0;
    int cur=0, completed=0;
    queue<int> q;
    bool inQueue[n];
    for(int i=0;i<n;i++) inQueue[i]=false;
    for(int i=0;i<n;i++)
        if(at[i]==0){ q.push(i); inQueue[i]=true; }
    while(completed < n){
        if(q.empty()){ cur++; continue; }
        int idx = q.front(); q.pop();
        int execTime = min(tq, rem[idx]);
        gs[gc]=cur; gantt[gc]=idx;
        cur += execTime; rem[idx] -= execTime;
        ge[gc]=cur; gc++;
        for(int i=0;i<n;i++)
            if(!inQueue[i] && at[i]<=cur && rem[i]>0)
                { q.push(i); inQueue[i]=true; }
        if(rem[idx]==0){
            ct[idx]=cur; tat[idx]=ct[idx]-at[idx];
            wt[idx]=tat[idx]-bt[idx]; completed++;
        } else { q.push(idx); }
    }
    cout<<"\n--- Gantt Chart ---\n|";
    for(int i=0;i<gc;i++) cout<<" "<<name[gantt[i]]<<" |";
    cout<<"\n"<<gs[0];
    for(int i=0;i<gc;i++) cout<<"--"<<ge[i];
    cout<<"\n";
    cout<<"\nProcess AT BT CT TAT WT\n";
    cout<<"--------------------------------\n";
    float ttat=0,twt=0;
    for(int i=0;i<n;i++){
        cout<<name[i]<<" "<<at[i]<<" "<<bt[i]<<" "
            <<ct[i]<<" "<<tat[i]<<" "<<wt[i]<<"\n";
        ttat+=tat[i]; twt+=wt[i];
    }
    cout<<"--------------------------------\n";
    cout<<"Average TAT : "<<ttat/n<<"\n";
    cout<<"Average WT  : "<<twt/n<<"\n";
    return 0;
}

 Question 9: MLQ - Multi-Level Queue
============================================================
Description:
 - Processes assigned to fixed queues by type.
 - High priority queue uses Round Robin (tq=2).
 - Low priority queue uses FCFS. High queue always served first.

Source Code (C++):

#include<string>
#include<queue>

int main(){
    cout << "============================" << endl;
    cout << " Mohd Anas " << endl;
    cout << " Roll : 24BCS061 " << endl;
    cout << " MLQ Scheduling " << endl;
    cout << " Q0=High(RR,tq=2) " << endl;
    cout << " Q1=Low (FCFS) " << endl;
    cout << "============================" << endl;
    int n;
    cout << "\nEnter number of processes: "; cin >> n;
    string name[n];

    bool done[n];

        cout<<"Enter Burst Time for P"<<(i+1)<<": "; cin>>bt[i];
        cout<<"Enter Queue Level for P"<<(i+1)<<" (0=High / 1=Low): "; cin>>ql[i];
        rem[i]=bt[i];

    int cur=0, completed=0, tq=2;
    bool inQueue[n];

    queue<int> q0, q1;
    for(int i=0;i<n;i++){
        if(at[i]==0){
            if(ql[i]==0){ q0.push(i); inQueue[i]=true; }
            else { q1.push(i); inQueue[i]=true; }
        }

    while(completed < n){
        for(int i=0;i<n;i++)
            if(!inQueue[i] && at[i]<=cur){
                if(ql[i]==0){ q0.push(i); inQueue[i]=true; }
                else { q1.push(i); inQueue[i]=true; }

        int idx=-1; bool isQ0=false;

        else if(!q1.empty()){ idx=q1.front(); q1.pop(); }

        gs[gc]=cur; gantt[gc]=idx;

        ge[gc]=cur; gc++;
        for(int i=0;i<n;i++)
            if(!inQueue[i] && at[i]<=cur){
                if(ql[i]==0){ q0.push(i); inQueue[i]=true; }
                else { q1.push(i); inQueue[i]=true; }

        if(rem[idx]==0){
            ct[idx]=cur; tat[idx]=ct[idx]-at[idx];

 Question 21: Banker's Algorithm
============================================================
Description:
 - Deadlock avoidance algorithm.
 - Checks if the system is in a safe state by finding a safe sequence.
 - Need = Max - Allocation.

Source Code (C++):

#include <iostream>
using namespace std;
int main() {
    cout << "Name : Mohd Anas\n";
    cout << "Roll : 24BCS061\n";
    cout << "Bankers Algorithm\n\n";
    int n, m;
    cout << "Enter number of processes: ";
    cin >> n;
    cout << "Enter number of resource types: ";
    cin >> m;
    int alloc[n][m], max[n][m], need[n][m];
    int avail[m];
    cout << "\nEnter Allocation Matrix:\n";
    for(int i = 0; i < n; i++)
        for(int j = 0; j < m; j++)
            cin >> alloc[i][j];
    cout << "\nEnter Max Matrix:\n";
    for(int i = 0; i < n; i++)
        for(int j = 0; j < m; j++)
            cin >> max[i][j];
    cout << "\nEnter Available Resources:\n";
    for(int i = 0; i < m; i++) cin >> avail[i];
    for(int i = 0; i < n; i++)
        for(int j = 0; j < m; j++)
            need[i][j] = max[i][j] - alloc[i][j];
    bool finish[n] = {false};
    int safeSeq[n];
    int work[m];
    for(int i = 0; i < m; i++) work[i] = avail[i];
    int count = 0;
    while(count < n) {
        bool found = false;
        for(int i = 0; i < n; i++) {
            if(!finish[i]) {
                bool canRun = true;
                for(int j = 0; j < m; j++) {
                    if(need[i][j] > work[j]) {
                        canRun = false;
                        break;
                    }
                }
                if(canRun) {
                    for(int j = 0; j < m; j++) work[j] += alloc[i][j];
                    safeSeq[count++] = i;
                    finish[i] = true;
                    found = true;
                }
            }
        }
        if(!found) {
            cout << "\nSystem is NOT in a safe state.\n";
            return 0;
        }
    }
    cout << "\nSystem is in a SAFE state.\nSafe sequence: ";
    for(int i = 0; i < n; i++) {
        cout << "P" << safeSeq[i];
        if(i != n - 1) cout << " -> ";
    }
    cout << endl;
    return 0;
}
